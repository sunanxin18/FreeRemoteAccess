cmake_minimum_required(VERSION 3.5.0)
project(freeremotedesk_ohos)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(FRD_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
if(CMAKE_BUILD_TYPE STREQUAL "Debug" OR CMAKE_BUILD_TYPE STREQUAL "debug")
    set(FRD_RUST_PROFILE "debug")
    set(FRD_GRAPHICS_POC_ENABLED 1)
else()
    set(FRD_RUST_PROFILE "release")
    set(FRD_GRAPHICS_POC_ENABLED 0)
endif()
if(NOT DEFINED FRD_OHOS_RUST_LIBRARY)
    set(FRD_OHOS_RUST_LIBRARY "${FRD_REPOSITORY_ROOT}/target/aarch64-unknown-linux-ohos/${FRD_RUST_PROFILE}/libfrd_shell_ohos_ffi.so")
endif()
if(NOT EXISTS "${FRD_OHOS_RUST_LIBRARY}")
    message(FATAL_ERROR "先运行 tools/build-harmony-hap.sh 构建 Rust 原生库")
endif()
add_library(frd_shell_ohos_ffi SHARED IMPORTED)
set_target_properties(frd_shell_ohos_ffi PROPERTIES IMPORTED_LOCATION "${FRD_OHOS_RUST_LIBRARY}")
add_library(entry SHARED napi_init.cpp app_bridge.cpp)
target_compile_definitions(entry PRIVATE FRD_GRAPHICS_POC_ENABLED=${FRD_GRAPHICS_POC_ENABLED})
target_include_directories(entry PRIVATE "${FRD_REPOSITORY_ROOT}/include")
target_compile_options(entry PRIVATE -Wall -Wextra -Werror -Wno-unused-command-line-argument)
target_link_libraries(entry PUBLIC frd_shell_ohos_ffi libace_napi.z.so libace_ndk.z.so libnative_window.so libhilog_ndk.z.so)
